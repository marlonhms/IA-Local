  <script>
document.querySelectorAll('[data-embed="software-delivery-agent-visual"]:not([data-wds-init])').forEach(function (root) {
  root.setAttribute('data-wds-init', '1');

  // Timing constants mirror the source verbatim: t starts at 350ms; each
  // step's 'running'→'success' duration is its own runMs; FLASH_MS is the
  // green success hold before the card border reverts to grey ('done').
  var STEP_RUN_MS = [2400, 2600, 3000];
  var FLASH_MS = 700;

  var cards = root.querySelectorAll('.software-delivery-agent-visual_card');
  var reasons = root.querySelectorAll('.software-delivery-agent-visual_reason');

  function setStatus(i, status) {
    var card = cards[i];
    var reason = reasons[i];
    if (!card || !reason) { return; }

    var runningBadge = card.querySelector('.software-delivery-agent-visual_badge--running');
    var successBadge = card.querySelector('.software-delivery-agent-visual_badge--success');
    var runningView = reason.querySelector('.software-delivery-agent-visual_reason-view--running');
    var doneView = reason.querySelector('.software-delivery-agent-visual_reason-view--done');

    card.classList.remove('is-running', 'is-success');
    if (runningBadge) { runningBadge.classList.remove('is-visible'); }
    if (successBadge) { successBadge.classList.remove('is-visible'); }
    if (runningView) { runningView.classList.remove('is-visible'); }
    if (doneView) { doneView.classList.remove('is-visible'); }

    if (status === 'pending') {
      reason.classList.remove('is-active');
      return;
    }

    reason.classList.add('is-active');

    if (status === 'running') {
      card.classList.add('is-running');
      if (runningBadge) { runningBadge.classList.add('is-visible'); }
      if (runningView) { runningView.classList.add('is-visible'); }
      return;
    }

    if (status === 'success') {
      card.classList.add('is-success');
      if (successBadge) { successBadge.classList.add('is-visible'); }
      if (doneView) { doneView.classList.add('is-visible'); }
      return;
    }

    // status === 'done' — border reverts to grey; the success chip and the
    // "Thought for Xs" reasoning view persist (showChip = status !== 'pending').
    if (successBadge) { successBadge.classList.add('is-visible'); }
    if (doneView) { doneView.classList.add('is-visible'); }
  }

  function runSequence() {
    var t = 350;
    for (var i = 0; i < STEP_RUN_MS.length; i++) {
      (function (idx, runAt, runMs) {
        setTimeout(function () { setStatus(idx, 'running'); }, runAt);
        setTimeout(function () { setStatus(idx, 'success'); }, runAt + runMs);
        setTimeout(function () { setStatus(idx, 'done'); }, runAt + runMs + FLASH_MS);
      })(i, t, STEP_RUN_MS[i]);
      // Next step's 'running' starts when the current step flips to 'success'
      // (its green flash overlaps the next step's start) — mirrors source.
      t = t + STEP_RUN_MS[i];
    }
  }

  var reduce = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;

  if (reduce) {
    // Reduced motion: all steps jump straight to 'done', no timers.
    for (var d = 0; d < cards.length; d++) { setStatus(d, 'done'); }
  } else if (typeof IntersectionObserver !== 'undefined') {
    var io = new IntersectionObserver(function (entries) {
      if (entries[0].isIntersecting) {
        runSequence();
        io.disconnect();
      }
    }, { threshold: 0.4 });
    io.observe(root);
  } else {
    runSequence();
  }

  // ---- Accordion toggle: the "Thought for Xs" heading expands the
  // reasoning text, pushing the result down (source: per-Reasoning `open`
  // state, toggled by the ChevronDown click handler). ----
  root.querySelectorAll('.software-delivery-agent-visual_reason-toggle').forEach(function (btn) {
    btn.addEventListener('click', function () {
      var wasOpen = btn.getAttribute('aria-expanded') === 'true';
      var next = !wasOpen;
      btn.setAttribute('aria-expanded', String(next));
      var chevron = btn.querySelector('.software-delivery-agent-visual_chevron');
      if (chevron) { chevron.classList.toggle('is-open', next); }
      var body = btn.parentElement;
      var collapsible = body ? body.querySelector('.software-delivery-agent-visual_collapsible') : null;
      if (collapsible) { collapsible.classList.toggle('is-open', next); }
    });
  });
});
  </script>
